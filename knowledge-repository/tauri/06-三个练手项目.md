## 项目一：TodoList 本地任务管理器（新手通关第 1 关）

### 1.1 它是啥

一个最小的、纯本地的待办清单。功能：
- 新建 / 编辑 / 删除 Todo
- 勾选完成 / 未完成
- 关键词搜索 + 状态过滤（全部 / 未完成 / 已完成）
- 关软件再打开，数据还在（用 store 插件持久化）
- 系统托盘：全局快捷键 Ctrl+Alt+T 快速显示主窗口
- 托盘菜单：今日未完成数量直接显示在 tooltip

**适合你刚学完 01 环境搭建 + 03 通信 + 04 store 权限，想写点真东西。**

### 1.2 技术栈

- 前端：Vue 3 + TypeScript + Pinia（或 React + Zustand 也行，本文给 Vue 代码）
- 持久化：官方 `tauri-plugin-store`（小数据，JSON 文件存 $APP 目录，不用 sqlite）
- 原生：托盘 + 全局快捷键
- 打包体积目标：Windows < 10MB

### 1.5 Rust 端完整代码（`src-tauri/src/lib.rs`）

```rust
use chrono::Local;
use serde::{Deserialize, Serialize};
use std::sync::Mutex;
use tauri::{App, AppHandle, Emitter, Manager, State};
use tauri_plugin_global_shortcut::{
    Code, GlobalShortcutExt, Modifiers, Shortcut, ShortcutState,
};
use tauri_plugin_store::{Store, StoreExt};
use tauri_plugin_tray::{
    MouseButton, MouseButtonState, TrayIconBuilder, TrayIconEvent,
};
use tauri_plugin_menu::{Menu, MenuItem, PredefinedMenuItem};

// ===== 数据结构 =====
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Todo {
    pub id: u64,
    pub title: String,
    pub done: bool,
    pub created_at: String,
    pub updated_at: String,
}

pub struct TodoState {
    pub store_key: String,
    pub seq: Mutex<u64>,
}

// ===== 读 store 里的列表 =====
fn load_todos(app: &AppHandle, key: &str) -> Result<Vec<Todo>, String> {
    let store = app
        .store_builder(key)
        .build()
        .map_err(|e| e.to_string())?;
    let todos: Vec<Todo> = store
        .get("todos")
        .unwrap_or_default();
    Ok(todos)
}

fn save_todos(app: &AppHandle, key: &str, todos: &[Todo]) -> Result<(), String> {
    let store = app
        .store_builder(key)
        .build()
        .map_err(|e| e.to_string())?;
    store
        .set("todos", serde_json::to_value(todos).unwrap())
        .map_err(|e| e.to_string())?;
    store.save().map_err(|e| e.to_string())?;
    Ok(())
}

fn refresh_tray_tooltip(app: &AppHandle, key: &str) {
    let count = load_todos(app, key).map(|ts| ts.iter().filter(|t| !t.done).count()).unwrap_or(0);
    if let Some(tray) = app.tray_by_id("todo-main-tray") {
        let _ = tray.set_tooltip(Some(format!("我的待办：还有 {} 件没做完", count)));
    }
}

// ===== 命令 =====
#[tauri::command]
pub fn list_todos(
    app: AppHandle,
    state: State<'_, TodoState>,
    keyword: Option<String>,
    done: Option<bool>,
) -> Result<Vec<Todo>, String> {
    let mut list = load_todos(&app, &state.store_key)?;
    // 排序：未完成在前，已完成在后；内部按创建时间倒序
    list.sort_by(|a, b| {
        a.done.cmp(&b.done)
            .then_with(|| b.created_at.cmp(&a.created_at))
    });
    // 过滤
    let kw = keyword.unwrap_or_default();
    list.retain(|t| {
        (kw.is_empty() || t.title.contains(&kw)) &&
        (done.is_none() || t.done == done.unwrap())
    });
    Ok(list)
}

#[tauri::command]
pub fn create_todo(
    app: AppHandle,
    state: State<'_, TodoState>,
    title: String,
) -> Result<Todo, String> {
    let title = title.trim().to_string();
    if title.is_empty() {
        return Err("标题不能为空".into());
    }
    let id = {
        let mut s = state.seq.lock().unwrap();
        *s += 1;
        *s
    };
    let now = Local::now().format("%Y-%m-%d %H:%M:%S").to_string();
    let todo = Todo {
        id,
        title,
        done: false,
        created_at: now.clone(),
        updated_at: now,
    };

    let mut list = load_todos(&app, &state.store_key)?;
    list.push(todo.clone());
    save_todos(&app, &state.store_key, &list)?;
    refresh_tray_tooltip(&app, &state.store_key);

    Ok(todo)
}

#[tauri::command]
pub fn update_todo(
    app: AppHandle,
    state: State<'_, TodoState>,
    id: u64,
    title: Option<String>,
    done: Option<bool>,
) -> Result<Todo, String> {
    let mut list = load_todos(&app, &state.store_key)?;
    let idx = list.iter().position(|t| t.id == id).ok_or("todo 不存在")?;
    let t = &mut list[idx];
    if let Some(new_title) = title {
        if new_title.trim().is_empty() {
            return Err("标题不能为空".into());
        }
        t.title = new_title.trim().to_string();
    }
    if let Some(new_done) = done {
        t.done = new_done;
    }
    t.updated_at = Local::now().format("%Y-%m-%d %H:%M:%S").to_string();
    let result = t.clone();
    save_todos(&app, &state.store_key, &list)?;
    refresh_tray_tooltip(&app, &state.store_key);
    Ok(result)
}

#[tauri::command]
pub fn delete_todo(
    app: AppHandle,
    state: State<'_, TodoState>,
    id: u64,
) -> Result<(), String> {
    let mut list = load_todos(&app, &state.store_key)?;
    let before = list.len();
    list.retain(|t| t.id != id);
    if list.len() == before {
        return Err("todo 不存在".into());
    }
    save_todos(&app, &state.store_key, &list)?;
    refresh_tray_tooltip(&app, &state.store_key);
    Ok(())
}

// ===== 入口 =====
fn setup_tray(app: &mut App) -> Result<(), Box<dyn std::error::Error>> {
    let handle = app.handle();

    let menu = Menu::with_items(handle, &[
        &MenuItem::with_id(handle, "show", "显示主窗口", true, Some("Ctrl+Alt+T"))?,
        &MenuItem::with_id(handle, "new", "新增一条待办", true, None::<&str>)?,
        &PredefinedMenuItem::separator(handle)?,
        &PredefinedMenuItem::quit(handle, Some("退出"))?,
    ])?;

    TrayIconBuilder::with_id("todo-main-tray")
        .icon(tauri::image::Image::from_bytes(include_bytes!("../icons/icon.png"))?)
        .menu(&menu)
        .tooltip("我的待办：还有 0 件没做完")
        .on_menu_event(|app, event| match event.id().0.as_str() {
            "show" => {
                if let Some(w) = app.get_webview_window("main") {
                    let _ = w.show();
                    let _ = w.unminimize();
                    let _ = w.set_focus();
                }
            }
            "new" => {
                let _ = app.emit("tray-new-todo", ());
                if let Some(w) = app.get_webview_window("main") {
                    let _ = w.show();
                    let _ = w.set_focus();
                }
            }
            _ => {}
        })
        .on_tray_icon_event(|tray, event| {
            if let TrayIconEvent::Click {
                button: MouseButton::Left,
                button_state: MouseButtonState::Up,
                ..
            } = event
            {
                let app = tray.app_handle();
                if let Some(w) = app.get_webview_window("main") {
                    let visible = w.is_visible().unwrap_or(false);
                    if visible { let _ = w.hide(); } else { let _ = w.show(); let _ = w.set_focus(); }
                }
            }
        })
        .build(app)?;

    Ok(())
}

fn setup_shortcut(app: &mut App) -> Result<(), Box<dyn std::error::Error>> {
    let handle = app.handle();
    let hotkey = Shortcut::new(Some(Modifiers::CONTROL | Modifiers::ALT), Code::KeyT);
    handle.global_shortcut().on_shortcut(move |h, sc, state| {
        if state != ShortcutState::Pressed { return; }
        if sc == &hotkey {
            if let Some(w) = h.get_webview_window("main") {
                let _ = w.show(); let _ = w.unminimize(); let _ = w.set_focus();
            }
        }
    });
    handle.global_shortcut().register(hotkey)?;
    Ok(())
}

#[cfg_attr(mobile, tauri::mobile_entry_point)]
pub fn run() {
    tauri::Builder::default()
        .plugin(tauri_plugin_store::Builder::new().build())
        .plugin(tauri_plugin_tray::init())
        .plugin(tauri_plugin_global_shortcut::Builder::new().build())
        .manage(TodoState {
            store_key: "todos.bin".into(),
            seq: Mutex::new(1_000_000),
        })
        .setup(|app| {
            setup_tray(app)?;
            setup_shortcut(app)?;
            refresh_tray_tooltip(&app.handle(), "todos.bin");
            Ok(())
        })
        .invoke_handler(tauri::generate_handler![
            list_todos, create_todo, update_todo, delete_todo
        ])
        .run(tauri::generate_context!())
        .expect("error running todo app");
}
```

```vue
<script setup lang="ts">
import { onMounted, onUnmounted, ref, computed } from 'vue'
import { invoke } from '@tauri-apps/api/core'
import { listen } from '@tauri-apps/api/event'
import { getCurrentWindow } from '@tauri-apps/api/window'

interface Todo {
  id: number
  title: string
  done: boolean
  created_at: string
  updated_at: string
}

type Filter = 'all' | 'active' | 'done'

const todos = ref<Todo[]>([])
const newTitle = ref('')
const keyword = ref('')
const filter = ref<Filter>('all')
const editingId = ref<number | null>(null)
const editingBuf = ref('')
let unlistenNew: Awaited<ReturnType<typeof listen>> | null = null

const visible = computed(() => {
  return todos.value.filter((t) => {
    if (filter.value === 'active' && t.done) return false
    if (filter.value === 'done' && !t.done) return false
    if (keyword.value && !t.title.includes(keyword.value)) return false
    return true
  })
})
const remainCount = computed(() => todos.value.filter((t) => !t.done).length)

const load = async () => {
  todos.value = await invoke<Todo[]>('list_todos', {
    keyword: keyword.value || null,
    done: filter.value === 'all' ? null : filter.value === 'done',
  })
}

const create = async () => {
  const title = newTitle.value.trim()
  if (!title) return
  await invoke('create_todo', { title })
  newTitle.value = ''
  await load()
}

const toggle = async (id: number) => {
  const t = todos.value.find((x) => x.id === id)
  if (!t) return
  await invoke('update_todo', { id, done: !t.done })
  await load()
}

const saveEdit = async (id: number) => {
  await invoke('update_todo', { id, title: editingBuf.value })
  editingId.value = null
  editingBuf.value = ''
  await load()
}

const del = async (id: number) => {
  if (!confirm('确定删除？')) return
  await invoke('delete_todo', { id })
  await load()
}

onMounted(async () => {
  await load()
  unlistenNew = await listen('tray-new-todo', () => {
    getCurrentWindow().setFocus()
    document.getElementById('new-input')?.focus()
  })
})
onUnmounted(() => unlistenNew?.())
</script>
```

## 项目二：本地笔记 Markdown 编辑器（通关第 2 关）

### 2.1 它是啥

一个「VSCode 一半」水平的本地 Markdown 编辑器：
- 左边目录树（用户选一个根目录）
- 中间编辑器（左右分栏：左边源码，右边实时预览）
- 创建 / 重命名 / 删除 / 保存文件和文件夹
- 导出单篇为 `.md` / `.html` / `.pdf`（PDF 用系统打印对话框）
- 多标签页、关闭前提示未保存
- 最近打开的 10 个目录，下次打开一键切

**你需要用到的知识：** fs 读/写文件 + dialog 选目录 + store 存最近目录 + 事件 + 多标签页状态管理。

### 2.2 技术栈

- 前端：Vue 3 + TS + Pinia + Markdown renderer（`marked`）+ 代码高亮（`highlight.js`）
- Rust 侧：fs 插件 + dialog 插件 + store 插件

### 2.3 完整最小实现（关键命令）

Rust 端 lib.rs 里加这 6 个命令，就够支撑整个编辑器了：

```rust
use std::fs;
use std::path::{Path, PathBuf};
use serde::{Serialize, Deserialize};
use tauri::{AppHandle, State};
use tauri_plugin_store::{Store, StoreExt};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FileNode {
    pub name: String,
    pub path: String,
    pub is_dir: bool,
    pub children: Option<Vec<FileNode>>,
}

fn build_tree(root: &Path) -> Result<FileNode, String> {
    let name = root.file_name()
        .map(|s| s.to_string_lossy().to_string())
        .unwrap_or_else(|| root.display().to_string());

    if !root.is_dir() {
        return Ok(FileNode {
            name,
            path: root.display().to_string(),
            is_dir: false,
            children: None,
        });
    }

    let mut children = Vec::new();
    for entry in fs::read_dir(root).map_err(|e| e.to_string())? {
        let entry = entry.map_err(|e| e.to_string())?;
        let p = entry.path();
        // 跳过隐藏文件和 node_modules
        let fname = p.file_name().map(|s| s.to_string_lossy().to_string()).unwrap_or_default();
        if fname.starts_with('.') || fname == "node_modules" || fname == ".git" {
            continue;
        }
        children.push(build_tree(&p)?);
    }
    children.sort_by(|a, b| a.is_dir.cmp(&b.is_dir).reverse().then_with(|| a.name.cmp(&b.name)));

    Ok(FileNode {
        name,
        path: root.display().to_string(),
        is_dir: true,
        children: Some(children),
    })
}

pub struct EditorState {
    pub recent_key: String,
}

// 命令 1：选根目录，读取目录树
#[tauri::command]
pub fn open_folder(path: String) -> Result<FileNode, String> {
    let p = PathBuf::from(&path);
    if !p.is_dir() {
        return Err("这不是一个目录".into());
    }
    build_tree(&p)
}

// 命令 2：存最近 10 个目录
#[tauri::command]
pub fn push_recent(app: AppHandle, state: State<'_, EditorState>, path: String) -> Result<(), String> {
    let store: Store = app.store_builder(&state.recent_key).build().map_err(|e| e.to_string())?;
    let mut recent: Vec<String> = store.get("recent").unwrap_or_default();
    recent.retain(|x| x != &path);
    recent.insert(0, path);
    recent.truncate(10);
    store.set("recent", serde_json::to_value(&recent).unwrap()).map_err(|e| e.to_string())?;
    store.save().map_err(|e| e.to_string())?;
    Ok(())
}

#[tauri::command]
pub fn list_recent(app: AppHandle, state: State<'_, EditorState>) -> Result<Vec<String>, String> {
    let store: Store = app.store_builder(&state.recent_key).build().map_err(|e| e.to_string())?;
    Ok(store.get("recent").unwrap_or_default())
}

// 命令 3：读 md 文件内容
#[tauri::command]
pub fn read_markdown(path: String) -> Result<String, String> {
    let p = Path::new(&path);
    if !p.is_file() {
        return Err("文件不存在".into());
    }
    let ext = p.extension().and_then(|e| e.to_str()).unwrap_or("").to_lowercase();
    if ext != "md" && ext != "markdown" && ext != "txt" {
        return Err(format!("不是 md 文件（{}）", ext));
    }
    fs::read_to_string(p).map_err(|e| e.to_string())
}

// 命令 4：写 md 文件
#[tauri::command]
pub fn write_markdown(path: String, content: String) -> Result<(), String> {
    let p = Path::new(&path);
    if let Some(parent) = p.parent() {
        fs::create_dir_all(parent).map_err(|e| e.to_string())?;
    }
    fs::write(p, content).map_err(|e| e.to_string())
}

// 命令 5：新建文件 / 文件夹
#[tauri::command]
pub fn create_entry(parent_dir: String, name: String, is_dir: bool) -> Result<(), String> {
    let base = PathBuf::from(&parent_dir);
    if !base.is_dir() { return Err("父目录不存在".into()); }
    if name.is_empty() { return Err("名字不能为空".into()); }
    if name.contains('/') || name.contains('\\') { return Err("不能包含路径分隔符".into()); }
    let target = base.join(&name);
    if target.exists() { return Err("这个名字已经有了".into()); }
    if is_dir {
        fs::create_dir_all(&target).map_err(|e| e.to_string())?;
    } else {
        fs::write(&target, "").map_err(|e| e.to_string())?;
    }
    Ok(())
}

// 命令 6：重命名 / 删除
#[tauri::command]
pub fn rename_entry(path: String, new_name: String) -> Result<(), String> {
    let p = PathBuf::from(&path);
    if !p.exists() { return Err("源不存在".into()); }
    if new_name.is_empty() || new_name.contains('/') || new_name.contains('\\') {
        return Err("名字不合法".into());
    }
    let target = p.parent().unwrap().join(&new_name);
    if target.exists() { return Err("目标已存在".into()); }
    fs::rename(&p, &target).map_err(|e| e.to_string())
}

#[tauri::command]
pub fn delete_entry(path: String, is_dir: bool) -> Result<(), String> {
    let p = PathBuf::from(&path);
    if !p.exists() { return Err("路径不存在".into()); }
    if is_dir {
        fs::remove_dir_all(&p).map_err(|e| e.to_string())?;
    } else {
        fs::remove_file(&p).map_err(|e| e.to_string())?;
    }
    Ok(())
}
```

#### ② Rust 后台线程每 2 秒 emit 一次系统状态

```rust
use sysinfo::{CpuExt, DiskExt, NetworkExt, System, SystemExt};
use tauri::{App, AppHandle, Manager, Emitter};
use serde::Serialize;

#[derive(Serialize)]
pub struct SystemSnapshot {
    cpu: f32,
    mem_percent: f32,
    mem_used_gb: f32,
    mem_total_gb: f32,
    disks: Vec<DiskSnap>,
    net_down_kbps: f32,
    net_up_kbps: f32,
    time_str: String,
}
#[derive(Serialize)]
pub struct DiskSnap {
    mount: String,
    used_percent: f32,
    free_gb: f32,
}

fn setup_system_monitor(app: &mut App) -> Result<(), Box<dyn std::error::Error>> {
    let h = app.handle().clone();
    let mut sys = System::new_all();

    std::thread::spawn(move || {
        let mut last_down: u64 = 0;
        let mut last_up: u64 = 0;
        let mut last_time = std::time::Instant::now();

        loop {
            std::thread::sleep(std::time::Duration::from_millis(2000));
            sys.refresh_all();

            // CPU 平均值
            let cpu_avg = sys.cpus().iter().map(|c| c.cpu_usage()).sum::<f32>()
                / sys.cpus().len() as f32;

            // 内存
            let total = sys.total_memory() as f32;
            let used = sys.used_memory() as f32;
            let mem_total_gb = total / 1024.0 / 1024.0 / 1024.0;
            let mem_used_gb = used / 1024.0 / 1024.0 / 1024.0;
            let mem_percent = used / total * 100.0;

            // 磁盘
            let disks = sys.disks().iter().map(|d| {
                let total = d.total_space() as f32;
                let avail = d.available_space() as f32;
                DiskSnap {
                    mount: d.mount_point().display().to_string(),
                    used_percent: (total - avail) / total * 100.0,
                    free_gb: avail / 1024.0 / 1024.0 / 1024.0,
                }
            }).collect::<Vec<_>>();

            // 网络（算每秒差值）
            let (down, up) = sys
                .networks()
                .iter()
                .fold((0u64, 0u64), |acc, (_, n)| {
                    (acc.0 + n.received(), acc.1 + n.transmitted())
                });
            let elapsed = last_time.elapsed().as_secs_f32().max(0.001);
            let down_kb = (down - last_down) as f32 / elapsed / 1024.0;
            let up_kb = (up - last_up) as f32 / elapsed / 1024.0;
            last_down = down;
            last_up = up;
            last_time = std::time::Instant::now();

            let snap = SystemSnapshot {
                cpu: cpu_avg,
                mem_percent,
                mem_used_gb,
                mem_total_gb,
                disks,
                net_down_kbps: down_kb.max(0.0),
                net_up_kbps: up_kb.max(0.0),
                time_str: chrono::Local::now().format("%H:%M:%S").to_string(),
            };

            let _ = h.emit("system-snapshot", snap);
        }
    });
    Ok(())
}
```

#### ④ 剪贴板历史（监听剪贴板变化）

```rust
use arboard::Clipboard;
use std::sync::Mutex;
use tauri::{Emitter, Manager};

pub struct ClipHistory {
    pub list: Mutex<Vec<String>>,
}

fn setup_clipboard_watcher(app: &mut App) -> Result<(), Box<dyn std::error::Error>> {
    let h = app.handle().clone();
    app.manage(ClipHistory { list: Mutex::new(Vec::new()) });

    std::thread::spawn(move || {
        let mut cb = Clipboard::new().expect("打开剪贴板失败");
        let mut last = String::new();
        loop {
            std::thread::sleep(std::time::Duration::from_millis(600));
            if let Ok(text) = cb.get_text() {
                let t = text.trim().to_string();
                if !t.is_empty() && t != last && t.len() < 10_000 {
                    last = t.clone();
                    let state = h.state::<ClipHistory>();
                    let mut list = state.list.lock().unwrap();
                    list.retain(|x| x != &t);
                    list.insert(0, t.clone());
                    list.truncate(50);
                    let _ = h.emit("clipboard-changed", list.clone());
                }
            }
        }
    });
    Ok(())
}

#[tauri::command]
pub fn copy_clipboard_item(state: State<'_, ClipHistory>, idx: usize) -> Result<(), String> {
    let mut cb = arboard::Clipboard::new().map_err(|e| e.to_string())?;
    let list = state.list.lock().unwrap();
    let item = list.get(idx).ok_or("索引超界")?;
    cb.set_text(item.clone()).map_err(|e| e.to_string())
}
```
